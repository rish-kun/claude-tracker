**Goal:** Build the Vite + React dashboards (user usage, admin by-login, admin by-account, admin settings) and the Vercel deployment, setup docs, and CI.

```ts
  onRequest({ request }) {

    return request

api.use(csrf)

/** Resolves to the response data or throws ApiError. */
export async function unwrap<R extends { data?: unknown; error?: unknown; response: Response }>(
  p: Promise<R>,
): Promise<NonNullable<R['data']>> {
  const { data, error, response } = await p
  if (!response.ok || error !== undefined) {
    const message = (error as { error?: string } | undefined)?.error ?? response.statusText
    throw new ApiError(response.status, message)
  }
  return data as NonNullable<R['data']>
}
```

```ts
export function formatTokens(n: number): string {
  const abs = Math.abs(n)
  if (abs >= 1e9) return `${trim(n / 1e9, 2)}B`
  if (abs >= 1e6) return `${trim(n / 1e6, 1)}M`
  if (abs >= 1e3) return `${trim(n / 1e3, 1)}k`
```

`apps/web/src/lib/range.ts`:
```ts
export type Preset = 'today' | '7d' | '30d' | '90d'

export interface Range {
  from: Date
  to: Date
  tz: string
}

const DAY_MS = 86_400_000
const PRESET_DAYS: Record<Preset, number> = { today: 1, '7d': 7, '30d': 30, '90d': 90 }

export function browserTz(): string {
  return Intl.DateTimeFormat().resolvedOptions().timeZone || 'UTC'
}

/** YYYY-MM-DD of `d` in `tz`. */
export function dayString(d: Date, tz: string): string {
  return new Intl.DateTimeFormat('en-CA', { timeZone: tz, year: 'numeric', month: '2-digit', day: '2-digit' }).format(d)
}

/** The instant at which the day containing `d` began in `tz`. */
export function startOfDay(d: Date, tz: string): Date {
  const parts = new Intl.DateTimeFormat('en-US', {
    timeZone: tz,
    hourCycle: 'h23',
    hour: '2-digit',
    minute: '2-digit',
    second: '2-digit',
  }).formatToParts(d)
  const get = (type: string) => Number(parts.find((p) => p.type === type)?.value ?? 0)
  const elapsed = ((get('hour') * 60 + get('minute')) * 60 + get('second')) * 1000 + d.getMilliseconds()
  return new Date(d.getTime() - elapsed)
}

export function presetRange(p: Preset, now = new Date(), tz = browserTz()): Range {
  const days = PRESET_DAYS[p]
  const from = startOfDay(new Date(now.getTime() - (days - 1) * DAY_MS), tz)
  return { from, to: new Date(now.getTime() + 60_000), tz }
}

/** Every calendar day in `tz` touched by the range, in order. */
export function datesBetween(r: Range): string[] {
  const out = new Set<string>()
  for (let t = r.from.getTime(); t < r.to.getTime(); t += DAY_MS / 2) {
    out.add(dayString(new Date(t), r.tz))
  }
  out.add(dayString(new Date(r.to.getTime() - 1), r.tz))
  return [...out].sort()
}

export function rangeQuery(r: Range) {
  return { from: r.from.toISOString(), to: r.to.toISOString(), tz: r.tz }
}

export function rangeKey(r: Range): string {
  return `${r.from.toISOString()}|${r.to.toISOString()}|${r.tz}`
}
```

Note on the DST test: the range ends at 12:01Z on Nov 3, which is 07:01 EST, so `2026-11-03` is the last day; the 12-hour step plus `Set` guarantees no day is skipped or duplicated around the 25-hour Nov 1.

```ts
import type { SeriesPoint } from '../api/client'

export interface PivotKey {
  key: string
  label: string
}

export interface Pivot {
  rows: Array<Record<string, string | number>>
  keys: PivotKey[]
}

/** Turns long-form series points into one row per date with a column per key. */
export function pivot(points: SeriesPoint[], dates: string[], value: 'cost_usd' | 'total_tokens'): Pivot {
  const totals = new Map<string, { label: string; total: number }>()
  const cells = new Map<string, number>()
  for (const p of points) {
    const t = totals.get(p.key) ?? { label: p.label, total: 0 }
    t.total += p[value]
    totals.set(p.key, t)
    const cell = `${p.date}\u0000${p.key}`
    cells.set(cell, (cells.get(cell) ?? 0) + p[value])
  }
  const keys = [...totals.entries()]
    .sort((a, b) => b[1].total - a[1].total)
    .map(([key, t]) => ({ key, label: t.label }))
  const allDates = [...new Set([...dates, ...points.map((p) => p.date)])].sort()
  const rows = allDates.map((date) => {
    const row: Record<string, string | number> = { date }
    for (const k of keys) row[k.key] = cells.get(`${date}\u0000${k.key}`) ?? 0
    return row
  })
  return { rows, keys }
}
```

```ts
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query'
import { api, unwrap } from './client'
import { type Range, rangeKey, rangeQuery } from '../lib/range'

export const PAGE_SIZE = 20

export function useMe() {
  return useQuery({ queryKey: ['me'], queryFn: () => unwrap(api.GET('/me')), retry: false })
}

export function useLogout() {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: () => api.POST('/auth/logout'),
    onSuccess: () => qc.clear(),
  })
}

export function useMyUsage(r: Range) {
  return useQuery({
    queryKey: ['me', 'usage', rangeKey(r)],
    queryFn: () => unwrap(api.GET('/me/usage', { params: { query: rangeQuery(r) } })),
  })
}

export function useMySessions(page: number) {
  return useQuery({
    queryKey: ['me', 'sessions', page],
    queryFn: () => unwrap(api.GET('/me/sessions', { params: { query: { limit: PAGE_SIZE, offset: page * PAGE_SIZE } } })),
    placeholderData: (prev) => prev,
  })
}
```

````markdown
# Setting up cctrack

cctrack has three parts: the server + dashboard (deployed once), the `cctrack` CLI
(installed on each developer machine), and the Claude Code plugin (enabled on each machine).

## 1. Google OAuth client

1. In Google Cloud Console → APIs & Services → Credentials, create an **OAuth client ID**
   of type **Web application**.
2. Authorised redirect URIs:
   - `https://<your-domain>/api/auth/google/callback`
   - `http://localhost:5173/api/auth/google/callback` (local development)
3. Note the client ID and secret. If you use Google Workspace, set the consent screen to
   **Internal** so only your organisation can sign in.

## 2. Database (Neon)

Create a Neon project (or connect Neon through the Vercel marketplace). You need:
- `DATABASE_URL` — the **pooled** connection string (used by the server).
- `DATABASE_URL_UNPOOLED` — the **direct** connection string (used for migrations,
  which take a session advisory lock).

## 3. Deploy to Vercel

1. Import this repository as a Vercel project. `vercel.json` defines two services:
   `/api/*` → the Go server in `apps/server`, everything else → the Vite app in `apps/web`.
2. Set environment variables:

| Variable | Example | Notes |
|---|---|---|
| `DATABASE_URL` | `postgres://…-pooler…/neondb?sslmode=require` | pooled |
| `DATABASE_URL_UNPOOLED` | `postgres://…/neondb?sslmode=require` | direct; migrations |
| `GOOGLE_CLIENT_ID` / `GOOGLE_CLIENT_SECRET` | | from step 1 |
| `SESSION_SECRET` | output of `openssl rand -base64 48` | ≥ 32 bytes |
| `PUBLIC_URL` | `https://cctrack.example.com` | no trailing slash |
| `ALLOWED_DOMAINS` | `example.com` | Google Workspace domains (checked via the `hd` claim) |
| `ALLOWED_EMAILS` | `contractor@gmail.com` | individual addresses, comma-separated |
| `ADMIN_EMAILS` | `you@example.com` | can see everyone's usage |
| `AUTO_MIGRATE` | `true` | run migrations on startup (or run `make migrate` yourself) |

3. Deploy, then open `PUBLIC_URL` and sign in.

## 4. Install the CLI (each developer)

```bash
git clone <this repo> && cd <repo>
cargo install --path apps/cli
cctrack login --server https://cctrack.example.com
```
On a machine without a browser, use `cctrack login --paste`. Check with `cctrack status`.

## 5. Enable the Claude Code plugin (each developer)

In Claude Code:
```
/plugin marketplace add <github-org>/<repo>
/plugin install cctrack@cctrack
```
From then on Claude Code refuses to work until `cctrack login` has succeeded, uploads
token usage after every turn, and attaches short summaries of completed work.

### Enforce it (recommended for shared subscriptions)

Admins can make the plugin mandatory with Claude Code **managed settings**
(`/Library/Application Support/ClaudeCode/managed-settings.json` on macOS,
`/etc/claude-code/managed-settings.json` on Linux):

```json
{
  "extraKnownMarketplaces": {
    "cctrack": { "source": { "source": "github", "repo": "<github-org>/<repo>" } }
  },
  "enabledPlugins": { "cctrack@cctrack": true },
  "allowManagedHooksOnly": true
}
```
`enabledPlugins` in managed settings force-enables the plugin so users cannot turn it
off, and `allowManagedHooksOnly` stops user-level hooks from overriding it (hooks from
force-enabled plugins are exempt, so cctrack's hooks still run).

## What is collected

Per assistant response: session id, message id, model, timestamp, token counts
(input, output, cache writes, cache reads), speed, and whether it came from a subagent;
per session: working directory, git branch, device hostname, and an optional 1–2 line
summary written by a Haiku subagent. Prompts, responses and file contents never leave
the machine. Plan-limit percentages come from Claude Code's cached utilisation in
````
